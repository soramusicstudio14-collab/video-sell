document.getElementById("year").textContent = new Date().getFullYear();

// --- Pricing table + plan select (₹11 / index 0 hamesha default) ---
const tbody = document.getElementById("planTable");
const select = document.getElementById("plan");
CONFIG.PLANS.forEach((p, i) => {
  tbody.innerHTML += `<tr><td class="price">₹${p.price}</td><td>${p.subs.toLocaleString("en-IN")}</td><td>${p.hours} ghante baad</td></tr>`;
  const selectedAttr = i === CONFIG.DEFAULT_PLAN_INDEX ? "selected" : "";
  select.innerHTML += `<option value="${i}" ${selectedAttr}>₹${p.price} — ${p.subs.toLocaleString("en-IN")} subscribers (${p.hours}h)</option>`;
});
select.value = String(CONFIG.DEFAULT_PLAN_INDEX);

function updateAmount() {
  const p = CONFIG.PLANS[select.value];
  document.getElementById("planAmount").textContent = "₹" + p.price;
}
select.addEventListener("change", updateAmount);
updateAmount();

// --- Reviews slider: list ko do baar render karte hain taaki loop seamless dikhe ---
const reviewTrack = document.getElementById("reviewTrack");
function starString(n) { return "★".repeat(n) + "☆".repeat(5 - n); }
function renderReviews() {
  const cards = CONFIG.REVIEWS.map(r => `
    <div class="review-card">
      <div class="review-stars">${starString(r.stars)}</div>
      <div class="review-text">"${r.text}"</div>
      <div class="review-name">— ${r.name}</div>
    </div>`).join("");
  reviewTrack.innerHTML = cards + cards; // duplicate for infinite scroll illusion
}
renderReviews();

// --- Order form ---
const statusEl = document.getElementById("status");
function showStatus(msg, ok) {
  statusEl.textContent = msg;
  statusEl.className = ok ? "ok" : "err";
}

document.getElementById("payBtn").addEventListener("click", async () => {
  const ytHandle = document.getElementById("ytHandle").value.trim();
  const email = document.getElementById("email").value.trim();
  const plan = CONFIG.PLANS[select.value];

  if (!ytHandle.startsWith("@") || ytHandle.length < 3) {
    return showStatus("Sahi YouTube handle bharein (jaise @yourchannel)", false);
  }
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    return showStatus("Sahi email ID bharein", false);
  }

  const btn = document.getElementById("payBtn");
  btn.disabled = true;
  btn.textContent = "Order ban raha hai...";

  try {
    // Step 1: Netlify function order create karta hai + Google Sheet me "pending" row daalta hai
    const res = await fetch(CONFIG.CREATE_ORDER_URL, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        ytHandle, email,
        planIndex: Number(select.value),
        amount: plan.price, subs: plan.subs, hours: plan.hours
      })
    });
    const data = await res.json();
    if (!data.orderId) throw new Error(data.error || "Order create nahi ho paya");

    // Step 2: Razorpay checkout kholein
    const rzp = new Razorpay({
      key: CONFIG.RAZORPAY_KEY_ID,
      amount: data.amount, // paise me
      currency: "INR",
      name: "Subscriber Gift Video",
      description: plan.subs.toLocaleString("en-IN") + " subscribers video",
      order_id: data.orderId,
      prefill: { email: email },
      handler: function (response) {
        showStatus("✅ Payment ho gaya! Aapka video " + plan.hours + " ghante me is email par aayega: " + email + ". Order ID: " + data.internalOrderId, true);
      },
      modal: {
        ondismiss: function () {
          btn.disabled = false;
          btn.textContent = "Pay & Order";
        }
      },
      theme: { color: "#ff5f6d" }
    });
    rzp.on("payment.failed", function (response) {
      showStatus("❌ Payment fail ho gaya: " + response.error.description, false);
      btn.disabled = false;
      btn.textContent = "Pay & Order";
    });
    rzp.open();
  } catch (e) {
    showStatus("❌ " + e.message, false);
    btn.disabled = false;
    btn.textContent = "Pay & Order";
  }
});
